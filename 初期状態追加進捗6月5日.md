function animate() {
  requestAnimationFrame(animate);
  // 再生中
  if (isPlaying) {
    currentTime += 0.016;
  }
  // UI更新
  document.getElementById("timeSlider").value =currentTime;
  document.getElementById("currentTime").textContent =currentTime.toFixed(1) + "s";
  // タイムライン位置
  const timelineX = currentTime * 50;
  document.getElementById("playhead").style.left =(150 + timelineX) + "px";
  document.getElementById("playheadLine").style.left =(150 + timelineX) + "px";
  // 再生中だけ自動スクロール
  if (isPlaying) {
    const ruler =document.getElementById("timelineRuler");
    const keyframeArea =document.getElementById("keyframeArea");
    // 赤線中央基準
    const scrollPosition =timelineX - 200;
    // 最大スクロール量
    const maxScroll =keyframeArea.scrollWidth -keyframeArea.clientWidth;
    // 範囲制限
    const scrollX = Math.min(
      Math.max(0, scrollPosition),maxScroll
    );
    // 上下同期
    ruler.scrollLeft = scrollX;
    keyframeArea.scrollLeft = scrollX;
  }
  applyKeyframes();
  renderer.render(scene, camera);
}

function syncTimelineScroll() {
  const ruler = document.getElementById("timelineRuler");
  const keyframeArea =document.getElementById("keyframeArea");
  let syncingFromRuler = false;
  let syncingFromKeyframe = false;
  // 上 → 下
  ruler.addEventListener("scroll", () => {
    if (syncingFromKeyframe) return;
    syncingFromRuler = true;
    keyframeArea.scrollLeft = ruler.scrollLeft;
    fixTrackLabelsPosition();
    syncingFromRuler = false;
  });
  // 下 → 上
  keyframeArea.addEventListener("scroll", () => {
    fixTrackLabelsPosition();
    if (syncingFromRuler) return;
    syncingFromKeyframe = true;
    ruler.scrollLeft = keyframeArea.scrollLeft;
    syncingFromKeyframe = false;
  });
}

function addKeyframe(event) {
  if (justDraggedKeyframe) {
    return;
  }
  const keyframeArea =document.getElementById("keyframeArea");
  // クリック位置取得
  const rect =keyframeArea.getBoundingClientRect();
  const labelWidth = 150;
  const x =event.clientX-rect.left+keyframeArea.scrollLeft-labelWidth;
  // 秒数変換
  const time = x / 50;
  // キーフレーム作成
  const keyframe =document.createElement("div");
  keyframe.className ="keyframe";
  // 位置
  keyframe.style.left =x + "px";
  const trackIndex = objects.indexOf(selectedObject);
  keyframe.style.top =(trackIndex * 50 + 20) + "px";
  // 情報保存
  keyframe.dataset.time =time;
  // 色保存
  keyframe.dataset.color =selectedObject.userData.lightColor;
  keyframe.dataset.isOn =selectedObject.userData.isOn;
  if (selectedObject.userData.type === "テープライト") {
    keyframe.dataset.rotation =selectedObject.rotation.y;
  }
  if (selectedObject.userData.type === "ムービングライト") {
    keyframe.dataset.rotation =selectedObject.userData.head.rotation.y;
  }
  keyframe.dataset.target =selectedObject.userData.name;
  // 表示
  keyframe.title =time.toFixed(1) + "s";
  document.getElementById("keyframeInner").appendChild(keyframe);
  // ドラッグ開始
  keyframe.addEventListener("mousedown",
   function(event) {
     draggingKeyframe = keyframe;
     movedKeyframe = false;
     // 前回キーフレーム選択解除
     if (selectedKeyframe) {
       selectedKeyframe.classList.remove("selectedKeyframe");
      }
     // 新しくキーフレーム選択
     selectedKeyframe = keyframe;
     selectedKeyframe.classList.add("selectedKeyframe");
     // 対象ライトも選択する
     selectObjectFromKeyframe(keyframe);
     // UI更新
     updateSelectedKeyframeUI();
     event.stopPropagation();
    }
  );
  keyframe.addEventListener("click", function(event) {
    event.stopPropagation();
  });
  keyframe.style.background =selectedObject.userData.lightColor;
   // イベント伝播停止
  event.stopPropagation();
  // 前回選択解除
  document
   .querySelectorAll(".keyframe")
   .forEach(kf => {kf.classList.remove("selectedKeyframe");
  });
  // 選択
  selectedKeyframe = keyframe;
  // 黄色枠追加
  keyframe.classList.add("selectedKeyframe");
  updateSelectedKeyframeUI();
}

function addSmartLight() {
  smartLightCount++;
  const movingLight = new THREE.Group();
  // 土台
  const base = new THREE.Mesh(
    new THREE.BoxGeometry(1, 0.5, 1),
    new THREE.MeshStandardMaterial({color: 0x222222})
  );
  movingLight.add(base);
  movingLight.userData.base = base;
  // ヘッド全体
  const yokeGroup = new THREE.Group();
  // ヘッドグループ
  const headGroup = new THREE.Group();
  // 半球
  const dome = new THREE.Mesh(
    new THREE.SphereGeometry(0.4,32,32,0,Math.PI * 2,0,Math.PI / 2),
    new THREE.MeshStandardMaterial({color: 0x111111})
  );
  // 半球を下向き
  dome.rotation.x = Math.PI;
  headGroup.add(dome);
  // フタ
  const cap = new THREE.Mesh(
    new THREE.CircleGeometry(0.4, 32),
    new THREE.MeshStandardMaterial({color: 0x555555})
  );
  // 少しだけズラす
  cap.position.y = -0.001;
  // 向き
  cap.rotation.x = -Math.PI / 2;
  headGroup.add(cap);
  // 光の線
  // 光の線（円錐台）
  const beam = new THREE.Mesh(
    new THREE.CylinderGeometry(
     0.38, // 上側（小さい円）
     1.2,  // 下側（大きい円）
     5,    // 長さ
     32
    ),
    new THREE.MeshBasicMaterial({color: 0xffff00,transparent: true,opacity: 0.4})
  );
  beam.geometry.translate(0, -2.5, 0);
  beam.position.y=-0.02;
  beam.position.z=0;
  beam.rotation.z = Math.PI;
  // 最初は消しておく
  beam.visible = false;
  // ヘッドに追加
  headGroup.add(beam);
  // ヘッド位置
  headGroup.position.y = 0.8;
  // 左斜め上
  headGroup.rotation.y = Math.PI / 4;
  headGroup.rotation.x = -Math.PI / 6;
  yokeGroup.add(headGroup);
  movingLight.add(yokeGroup);
  // 全体位置
  movingLight.position.set(Math.random() * 4 - 2,0,Math.random() * 4 - 2);
  // 保存
  movingLight.userData.head = yokeGroup;
  movingLight.userData.cap = cap;
  movingLight.userData.beam = beam;
  movingLight.userData.type = "ムービングライト";
  movingLight.userData.name ="ムービングライト" +smartLightCount;

function addTapeLight() {
  tapeLightCount++;
  // テープライト全体
  const tapeGroup = new THREE.Group();
  // 黒い本体
  const body = new THREE.Mesh(
    new THREE.BoxGeometry(2, 0.2, 0.2),

  const lightSurface = new THREE.Mesh(

  objects.push(tapeGroup);
  addTrackLabel(tapeGroup.userData.name,objects.length - 1);

// マウス操作
function onMouseDown(event) {
  event.preventDefault();
  isDragging = true;
  mouse.x =(event.offsetX / renderer.domElement.clientWidth) * 2 - 1;
  mouse.y =-(event.offsetY / renderer.domElement.clientHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);
  const intersects =raycaster.intersectObjects(objects, true);
  // 何もクリックしてない
  if (intersects.length === 0) {
    // 前回選択解除
    if (selectedObject) {removeOutline(selectedObject);}
    selectedObject = null;
    showDefaultUI();
    return;
  }
  // クリック対象取得
  const clickedObject = intersects[0].object;
  let newSelectedObject = clickedObject;
  // 一番上のGroupまでたどる
  while (
    newSelectedObject.parent &&newSelectedObject.parent !== scene
  ) {
    newSelectedObject =newSelectedObject.parent;
  }
  // 同じオブジェクトなら維持
  if (selectedObject === newSelectedObject) {return;}
  // 前回選択解除
  if (selectedObject) {removeOutline(selectedObject);}
  // 新しく選択
  selectedObject = newSelectedObject;
  // キーフレーム編集終了
  isEditingKeyframe = false;
  // キーフレーム選択解除
  if (selectedKeyframe) {selectedKeyframe.classList.remove("selectedKeyframe");}
  selectedKeyframe = null;
  updateSelectedKeyframeUI();
  showObjectUI(selectedObject);
  addOutline(selectedObject);
}

function onMouseUp() {
  isDragging = false;
  draggingKeyframe = null;
  if (movedKeyframe) {
    justDraggedKeyframe = true;
    movedKeyframe = false;
    setTimeout(() => {
      justDraggedKeyframe = false;
    }, 50);
  }
}

function onMouseWheel(event) {
  // 3D画面以外なら無視
  if (!renderer.domElement.contains(event.target) ) {return;}
  event.preventDefault();
  // カメラモードだけ
  if (controlMode !== "camera") return;
  cameraDistance += event.deltaY * 0.01;
  cameraDistance = Math.max(2,Math.min(30, cameraDistance));
}

function onMouseMove(event) {
  // キーフレームドラッグ
  if (draggingKeyframe) {
    const keyframeArea =document.getElementById("keyframeArea");
    const rect =keyframeArea.getBoundingClientRect();
    let x =event.clientX-rect.left-150+keyframeArea.scrollLeft;
    if (x < 0) x = 0;
    if (x > 3000) x = 3000;
    const oldX =parseFloat(draggingKeyframe.style.left);
    if (Math.abs(x - oldX) > 3) {
      movedKeyframe = true;
    }
    draggingKeyframe.style.left = x + "px";
    const time = x / 50;
    draggingKeyframe.dataset.time = time;
    draggingKeyframe.title =time.toFixed(1) + "s";
    updateSelectedKeyframeUI();
    return;
  }
  if (!isDragging) return;
  // カメラモード
  if (controlMode === "camera") {
    cameraYaw -= event.movementX * 0.005;
    // 上下回転
    cameraPitch -= event.movementY * 0.005;
    // 真上行き過ぎ防止
    const limit = Math.PI / 2 - 0.1;
    cameraPitch = Math.max(-limit,Math.min(limit, cameraPitch));
    // カメラ位置計算
    camera.position.x =cameraDistance *Math.cos(cameraPitch) *Math.sin(cameraYaw);
    camera.position.y =cameraDistance *Math.sin(cameraPitch);
    camera.position.z =cameraDistance *Math.cos(cameraPitch) *Math.cos(cameraYaw);
    // 中央を見る
    camera.lookAt(0,0,0);
    return;
  }
  // オブジェクトモード
  if (controlMode === "object") {
    if (!selectedObject) return;
    mouse.x =(event.offsetX /renderer.domElement.clientWidth)* 2 - 1;
    mouse.y =-(event.offsetY /renderer.domElement.clientHeight)* 2 + 1;
    raycaster.setFromCamera(mouse, camera);
    const plane =
      new THREE.Plane(
        new THREE.Vector3(0, 1, 0),
        0
      );
    const point = new THREE.Vector3();
    raycaster.ray.intersectPlane(
      plane,
      point
    );
    // 移動先座標
    let x = point.x;
    let z = point.z;
    // 教室の壁を超えないよう制限
    x = Math.max(-9, Math.min(9, x));
    z = Math.max(-9, Math.min(9, z));
    // 配置
    selectedObject.position.set(x,0.5,z);
    updatePositionUI(selectedObject);
  }
}

function deleteSelectedObject() {
  if (!selectedObject) return;
  // 名前取得
  const name = selectedObject.userData.name;
  // ムービングライト番号取得
  if (selectedObject.userData.type ==="ムービングライト") {
    const match = name.match(/\d+/);
    if (match) {
      const deletedNumber =Number(match[0]);
      // 最後の番号なら戻す
      if (deletedNumber === smartLightCount) {
        smartLightCount--;
      }
    }
  }
  // テープライト番号取得
  if (selectedObject.userData.type ==="テープライト") {
    const match = name.match(/\d+/);
    if (match) {
      const deletedNumber =Number(match[0]);
      // 最後の番号なら戻す
      if (deletedNumber === tapeLightCount) {
        tapeLightCount--;
      }
    }
  }
  // シーンから削除
  scene.remove(selectedObject);
  // 配列から削除
  objects = objects.filter(obj => obj !== selectedObject);
  // 選択解除
  selectedObject = null;
  // UI戻す
  showDefaultUI();
}

function addOutline(obj) {
  if (obj.userData.outline) return;
  let targetMesh = null;
  // ムービングライト
  if (obj.userData.type === "ムービングライト") {
    targetMesh = obj.userData.base;
  }
  // テープライト
  else if (obj.userData.type === "テープライト") {
    targetMesh = obj.userData.body;
  }
  // Meshじゃないなら終了
  if (!targetMesh) return;
  const edges = new THREE.EdgesGeometry(
    targetMesh.geometry
  );
  const line = new THREE.LineSegments(
    edges,
    new THREE.LineBasicMaterial({color: 0xffff00})
  );
  line.raycast = () => {};
  line.scale.set(1.05, 1.05, 1.05);
  targetMesh.add(line);
  obj.userData.outline = line;
}

function applyKeyframes() {
  // 全キーフレーム取得
  const keyframes = document.querySelectorAll(".keyframe");
  keyframes.forEach(keyframe => {
    // 保存時間
    const keyTime = Number(keyframe.dataset.time);
    // 保存色
    const color = keyframe.dataset.color;
    const isOn =keyframe.dataset.isOn === "true";
    const rotation =Number(keyframe.dataset.rotation);
    // 保存対象ライト名
    const targetName = keyframe.dataset.target;
    // 時間一致
    if (Math.abs(currentTime - keyTime) < 0.1) {
      objects.forEach(obj => {
        // 対象ライトだけ実行
        if (obj.userData.name !== targetName) return;
        // テープライト
        if (obj.userData.type === "テープライト") {
          // 向き適用
          obj.rotation.y = rotation;
          // ON/OFF適用
          obj.userData.isOn = isOn;
          if (isOn) {
            obj.userData.lightSurface.material.color.set(color);
            obj.userData.glow.material.color.set(color);
            obj.userData.glow.visible = true;
            const keyId =targetName + "_" +keyTime.toFixed(1);
            if (lastHueSentKeyframe !== keyId) {
              sendHueColor(color);
              lastHueSentKeyframe = keyId;
            }
          }
          else {
            obj.userData.lightSurface.material.color.set(0x555555);
            obj.userData.glow.visible = false;
            const keyId =targetName + "_" +keyTime.toFixed(1);
            if (lastHueSentKeyframe !== keyId) {
              sendHuePower(false);
              lastHueSentKeyframe = keyId;
            }
          }
        }
        // ムービングライト
        if (obj.userData.type === "ムービングライト") {
          // 向き適用
          obj.userData.head.rotation.y = rotation;
          // ON/OFF適用
          obj.userData.isOn = isOn;
          if (isOn) {
            obj.userData.cap.material.color.set(color);
            obj.userData.beam.material.color.set(color);
            obj.userData.beam.visible = true;
          }
          else {
            obj.userData.cap.material.color.set(0x555555);
            obj.userData.beam.visible = false;
          }
        }
      });
    }
  });
}

function selectObjectFromKeyframe(keyframe) {
  const targetName = keyframe.dataset.target;
  const targetObject = objects.find(obj =>
    obj.userData.name === targetName
  );
  if (!targetObject) return;
  // 前の黄色枠を消す
  if (selectedObject && selectedObject !== targetObject) {
    removeOutline(selectedObject);
  }
  // 選択中ライトを変更
  selectedObject = targetObject;

  showObjectUI(selectedObject);

  addOutline(selectedObject);
  // キーフレームの状態を3D画面とUIに反映
  applyKeyframeStateToObject(keyframe, selectedObject);
}

    } else {
      obj.userData.lightSurface.material.color.set(0x555555);
      obj.userData.glow.visible = false;
    }
    updateRotationUI();
  }
  // ムービングライト
  else if (obj.userData.type === "ムービングライト") {
    obj.userData.head.rotation.y = rotation;
    if (isOn) {
      obj.userData.cap.material.color.set(color);
      obj.userData.beam.material.color.set(color);
      obj.userData.beam.visible = true;
    } else {
      obj.userData.cap.material.color.set(0x555555);
      obj.userData.beam.visible = false;
    }
    updateMovingRotationUI();
  }
  document.getElementById("colorPicker").value = color;
  document.getElementById("powerButton").textContent =
    isOn ? "ON" : "OFF";
}

function updateSelectedKeyframeUI() {
  // 選択なし
  if (!selectedKeyframe) {
    document.getElementById("selectedKeyframeText").textContent ="選択中キーフレーム: なし";
    return;
  }
  // 時間取得
  const time = Number(selectedKeyframe.dataset.time);
  // UI更新
  document.getElementById("selectedKeyframeText"
  ).textContent ="選択中キーフレーム: " +time.toFixed(1) +"秒";
  // 色
  document.getElementById("colorPicker").value =selectedKeyframe.dataset.color;
  // 電源
  document.getElementById("powerButton").textContent =
  selectedKeyframe.dataset.isOn === "true"? "ON": "OFF";
  if (selectedObject &&selectedObject.userData.type === "テープライト") {
    const degree =THREE.MathUtils.radToDeg(Number(selectedKeyframe.dataset.rotation));
    document.getElementById("rotationSlider").value = degree;
    document.getElementById("rotationValue").textContent =Math.round(degree) + "°";
  }
  if (selectedObject &&selectedObject.userData.type === "ムービングライト") {
    const degree =THREE.MathUtils.radToDeg(Number(selectedKeyframe.dataset.rotation));
    document.getElementById("movingRotationSlider").value = degree;
    document.getElementById("movingRotationValue").textContent =Math.round(degree) + "°";
  }
}

function removeOutline(obj) {
  if (!obj.userData.outline) return;
  obj.userData.outline.parent.remove(obj.userData.outline);
  obj.userData.outline.geometry.dispose();
  obj.userData.outline.material.dispose();
  obj.userData.outline = null;
}

function loadDefaultScene() {
  // ムービングライト2台
  addSmartLight();
  addSmartLight();
  // テープライト2本
  addTapeLight();
  addTapeLight();
  // 位置調整
  objects[0].position.set(-3, 0, -3);
  objects[1].position.set(3, 0, -3);
  objects[2].position.set(-3, 0.5, 3);
  objects[3].position.set(3, 0.5, 3);
}

function resetScene() {
  if (!confirm("初期状態に戻しますか？")) {
    return;
  }
  // ライト削除
  objects.forEach(obj => {scene.remove(obj);});
  // 配列を空に
  objects = [];
  // 選択解除
  selectedObject = null;
  selectedKeyframe = null;
  // 番号リセット
  smartLightCount = 0;
  tapeLightCount = 0;
  // トラック名削除
  document.getElementById("trackLabels").innerHTML = "";
  // キーフレーム削除
  document.querySelectorAll(".keyframe").forEach(kf => {kf.remove();});
  // UI戻す
  showDefaultUI();
  // 初期ライト生成
  loadDefaultScene();
}
